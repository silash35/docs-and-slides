% --- Elementos pós-textuais ---
\newcounter{apendice}
\newcounter{anexo}

% Formato das referências
\renewcommand{\theapendice}{\Alph{apendice}}
\renewcommand{\theanexo}{\Alph{anexo}}

% Comando para apêndices
\newcommand{\apendice}[1]{%
  \refstepcounter{apendice}%
  \section*{APÊNDICE \theapendice\ - #1}%
  \addcontentsline{toc}{section}{APÊNDICE \theapendice\ - #1}%
}

% Comando para anexos
\newcommand{\anexo}[1]{%
  \refstepcounter{anexo}%
  \section*{ANEXO \theanexo\ - #1}%
  \addcontentsline{toc}{section}{ANEXO \theanexo\ - #1}%
}
