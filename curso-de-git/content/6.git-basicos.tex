\section{Prática com Git}

\begin{frame}{Nosso projeto}

  \begin{block}{Objetivo}
    Vamos escrever o cardápio de um restaurante utilizando o Git para acompanhar o processo de desenvolvimento.
  \end{block}
  \begin{columns}[T]
    \begin{column}{0.48\textwidth}
      \begin{block}{Ambiente Local}
        No seu computador:
        \begin{itemize}
          \item Instale o Git pelo site oficial.
          \item Crie uma conta no GitHub.
        \end{itemize}
        \vspace{0.2cm}
        \centering

        \href{https://git-scm.com/}{\textbf{🌐 git-scm.com}} \quad \href{https://github.com/}{\textbf{🌐 github.com}}
      \end{block}
    \end{column}
    \begin{column}{0.48\textwidth}
      \begin{block}{Ambiente Online}
        Vamos utilizar o GitHub Codespaces.

        \vspace{0.2cm}
        Ele fornece um ambiente de desenvolvimento já configurado para trabalharmos com o Git.

        \vspace{0.2cm}
        \centering
        \href{https://github.com/codespaces}{\textbf{🌐 github.com/codespaces}}
      \end{block}
    \end{column}
  \end{columns}
\end{frame}

\begin{frame}{Primeiro passo: identificar o autor}
  \begin{block}{Configurando o Git}
    Antes de começar, vamos informar quem está fazendo os commits.

    \vspace{0.3cm}

    \texttt{git config --global user.name "Your Name"}

    \texttt{git config --global user.email "you@example.com"}
  \end{block}
  No \textbf{GitHub Codespaces}, essa configuração já vem preparada para você.

  \end{alertblock}
\end{frame}

\begin{frame}{Criando o projeto}
  \begin{block}{Opção 1: criando localmente}
    Podemos começar criando um repositório na nossa máquina:
    \vspace{0.3cm} \\
    \texttt{git init}
    \vspace{0.3cm} \\
    Depois de criar o repositório no GitHub, adicionamos o repositório remoto:
    \vspace{0.3cm} \\
    \texttt{git remote add origin "URL do repositório"}
  \end{block}
  \begin{block}{Opção 2: repositório já existente}
    Se o repositório já foi criado no GitHub, podemos cloná-lo:
    \vspace{0.3cm} \\
    \texttt{git clone "URL do repositório"}
  \end{block}
  O \textbf{GitHub Codespaces} já realiza o clone do repositório automaticamente.
\end{frame}

\begin{frame}{Nosso primeiro commit}
  \begin{columns}
    \column{0.5\textwidth}

    \begin{block}{Criando o arquivo README.md}
      \texttt{touch README.md}
    \end{block}

    \begin{block}{Verificando as alterações}
      \texttt{git status}
    \end{block}

    \begin{block}{Preparando o commit}
      \texttt{git add .}
    \end{block}

    \begin{block}{Registrando o primeiro estado do projeto}
      \texttt{git commit -m "Add README.md file"}
    \end{block}

    \begin{block}{Subindo para o repositório remoto}
      \texttt{git push}
    \end{block}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=0.75\linewidth]{commit-meme.png}
  \end{columns}
\end{frame}


\begin{frame}{Consultando a história do projeto}

  \begin{columns}
    \column{0.5\textwidth}
    \begin{block}{\texttt{git log}: histórico de commits}
      \texttt{git log}
      \vspace{0.2cm} \\
      Podemos ver como o projeto foi evoluindo ao longo do tempo.
    \end{block}
    \begin{block}{\texttt{git show}: detalhes de um commit}
      \texttt{git show "commit"}
      \vspace{0.2cm}
      Mostra as informações e as alterações realizadas em um commit específico.
    \end{block}
    \column{0.5\textwidth}
    \begin{block}{\texttt{git diff}: diferenças entre estados do projeto}
      \texttt{git diff "commit1" \ "commit2"}
      \vspace{0.2cm} \\
      Mostra as diferenças entre dois estados do projeto.
    \end{block}
    \begin{block}{\texttt{git blame}: autoria das linhas}
      \texttt{git blame "arquivo"}
      \vspace{0.2cm} \\
      Mostra, linha por linha, qual commit e qual autor realizou a última alteração.
    \end{block}
  \end{columns}
\end{frame}

\begin{frame}{Arquivos úteis}
  \begin{columns}
    \column{0.5\textwidth}
    \begin{itemize}
      \item \texttt{.gitignore}: listar arquivos que não devem ser versionados
            \begin{itemize}
              \item \texttt{arquivo.txt}: ignora um arquivo específico
              \item \texttt{*.log}: ignora todos os arquivos com determinada extensão
              \item \texttt{pasta/}: ignora uma pasta inteira
              \item \texttt{!arquivo.log}: não ignora esse arquivo, mesmo que uma regra anterior o ignore
            \end{itemize}
      \item \texttt{.gitkeep}: convenção para manter diretórios vazios no repositório
    \end{itemize}
    \column{0.5\textwidth}
    \centering
    \includegraphics[width=0.8\linewidth]{gitignore.jpeg}
  \end{columns}
\end{frame}
