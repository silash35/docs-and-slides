\section{Comandos Básicos do Git}

\begin{frame}{Fluxo mínimo para começar}
  \begin{enumerate}
    \item \texttt{git init} — inicializar repositório local
    \item \texttt{git clone <url>} — clonar repositório remoto
    \item \texttt{git status} — checar estado do repositório
    \item \texttt{git add <arquivo>} — preparar (stage) alterações
    \item \texttt{git commit -m "mensagem"} — criar um commit
    \item \texttt{git push} / \texttt{git pull} — sincronizar com remoto
  \end{enumerate}
\end{frame}

\begin{frame}[fragile]{Explorando o histórico}
  \begin{itemize}
    \item \texttt{git log} — histórico de commits (use \texttt{--oneline --graph --all})
    \item \texttt{git show <commit>} — mostrar conteúdo e metadados de um commit
    \item \texttt{git diff} — ver diferenças entre working tree, index e commits
  \end{itemize}
\end{frame}

\begin{frame}{Branches e fluxo de trabalho}
  \begin{itemize}
    \item \texttt{git branch} — listar/ criar branches
    \item \texttt{git checkout <branch>} ou \texttt{git switch <branch>} — trocar de branch
    \item \texttt{git merge <branch>} — mesclar branch no atual
    \item Fluxos comuns: feature branches, Git Flow, GitHub Flow (pull requests)
  \end{itemize}
\end{frame}

\begin{frame}[fragile]{Recuperação e correção}
  \begin{itemize}
    \item \texttt{git restore --staged <file>} — remover do stage
    \item \texttt{git reset --soft/--mixed/--hard <commit>} — resetar HEAD e index
    \item \texttt{git revert <commit>} — criar novo commit que desfaz outro commit (seguro em branches compartilhadas)
    \item \texttt{git rebase} — reaplicar commits em novo ponto base (atenção ao rebase de commits já compartilhados)
  \end{itemize}
\end{frame}

\begin{frame}{Arquivos úteis}
  \begin{itemize}
    \item \texttt{.gitignore} — listar arquivos que não devem ser versionados
    \item \texttt{.gitattributes} — controlar tratamento de arquivos, diffs, merges
  \end{itemize}
\end{frame}

\begin{frame}{Próximos passos sugeridos}
  \begin{itemize}
    \item Prática: criar repositórios, branches e PRs em uma plataforma como GitHub.
    \item Aprender integração contínua (CI) e proteção de branches.
    \item Avançar em temas: submodules, hooks, repositórios monorepo, segurança de commits.
  \end{itemize}
\end{frame}
